{{- /* Project write-ups are articles. Only explicit scholarly records use ScholarlyArticle; do not infer SoftwareApplication from a blog section. */ -}} {{- if and .IsPage (in site.Params.mainSections .Section) -}} {{- $desc := (partial "seo/metadata.html" .).description | htmlUnescape -}} {{- $author := site.Params.metadata.author -}} {{- with .Params.author -}} {{- $author = . -}} {{- end -}} {{- $personID := hugo.Data.entity.id -}} {{- $authorNode := dict "@type" "Person" "name" $author -}} {{- if eq $author hugo.Data.entity.name -}} {{- $authorNode = merge $authorNode (dict "@id" $personID "url" ("about/" | absURL)) -}} {{- end -}} {{- /* subsection = 2nd path segment of the permalink */ -}} {{- $segs := split (strings.Trim .RelPermalink "/") "/" -}} {{- $sub := "" -}} {{- if ge (len $segs) 2 -}} {{- $sub = index $segs 1 -}} {{- end -}} {{- /* Match BOTH plural and singular. The real sections are plural (/blog/publications/, /blog/tech-projects/) while this only tested the singular forms, so the mapping never fired and every post — including the peer-reviewed article — fell back to BlogPosting. The singular forms are kept because each post also carries a singular alias. */ -}} {{- $type := "BlogPosting" -}} {{- if and .Params.doi (in (slice "publications" "publication") $sub) -}} {{- $type = "ScholarlyArticle" -}} {{- end -}} {{- with .Params.schema_type -}} {{- $type = . -}} {{- end -}} {{- $publisher := dict "@type" "Person" "@id" $personID "name" site.Params.metadata.author "url" site.BaseURL -}} {{- if eq $type "ScholarlyArticle" -}} {{- with .Params.publisher -}} {{- $publisher = dict "@type" "Organization" "name" . -}} {{- end -}} {{- end -}} {{- /* properties common to every type */ -}} {{- $schema := dict "@context" "https://schema.org" "@type" $type "description" $desc "url" .Permalink "author" $authorNode "publisher" $publisher "@id" (printf "%s#article" .Permalink) "mainEntityOfPage" (dict "@type" "WebPage" "@id" (printf "%s#webpage" .Permalink)) "inLanguage" (partial "seo/language.html" .).tag -}} {{- if not .Date.IsZero -}} {{- $schema = merge $schema (dict "datePublished" (.Date.Format "2006-01-02T15:04:05-07:00")) -}} {{- end -}} {{- if not .Lastmod.IsZero -}} {{- $schema = merge $schema (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05-07:00")) -}} {{- end -}} {{- /* type-specific properties */ -}} {{- if eq $type "SoftwareApplication" -}} {{- $schema = merge $schema (dict "name" .Title "applicationCategory" (.Params.app_category | default "DeveloperApplication")) -}} {{- with .Params.operating_system -}} {{- $schema = merge $schema (dict "operatingSystem" .) -}} {{- end -}} {{- else -}} {{- $schema = merge $schema (dict "headline" .Title) -}} {{- /* The DOI identifies the full paper, not this site's shorter summary. */ -}} {{- if ne $type "ScholarlyArticle" -}} {{- $schema = merge $schema (dict "wordCount" .WordCount) -}} {{- end -}} {{- with .CurrentSection -}} {{- $schema = merge $schema (dict "articleSection" .Title) -}} {{- end -}} {{- end -}} {{- /* Bibliographic detail for real journal articles. Driven by front matter, so only posts that actually carry a DOI get these — the DOI is what lets Crossref, OpenAlex and Google Scholar reconcile this page with the published record rather than treating it as an unrelated blog post. */ -}} {{- if eq $type "ScholarlyArticle" -}} {{- with .Params.doi -}} {{- $schema = merge $schema (dict "@id" (printf "https://doi.org/%s" .) "identifier" (dict "@type" "PropertyValue" "propertyID" "DOI" "value" .) "sameAs" (printf "https://doi.org/%s" .)) -}} {{- end -}} {{- with .Params.authors -}} {{- $authors := slice -}} {{- range . -}} {{- $node := dict "@type" "Person" "name" . -}} {{- if eq . hugo.Data.entity.name -}} {{- $node = merge $node (dict "@id" hugo.Data.entity.id "url" ("about/" | absURL)) -}} {{- end -}} {{- $authors = $authors | append $node -}} {{- end -}} {{- /* Author order is meaningful: he is second of four, never sole author. */ -}} {{- $schema = merge $schema (dict "author" $authors) -}} {{- end -}} {{- with .Params.journal -}} {{- $periodical := dict "@type" "Periodical" "name" . -}} {{- with $.Params.publisher -}} {{- $periodical = merge $periodical (dict "publisher" (dict "@type" "Organization" "name" .)) -}} {{- end -}} {{- $container := $periodical -}} {{- with $.Params.volume -}} {{- $container = dict "@type" "PublicationVolume" "volumeNumber" . "isPartOf" $container -}} {{- end -}} {{- with $.Params.issue -}} {{- $container = dict "@type" "PublicationIssue" "issueNumber" . "isPartOf" $container -}} {{- end -}} {{- $schema = merge $schema (dict "isPartOf" $container) -}} {{- end -}} {{- with .Params.article_number -}} {{- $schema = merge $schema (dict "pagination" .) -}} {{- end -}} {{- end -}} {{- /* image (all types) */ -}} {{- with .Params.image -}} {{- $imgURL := . | absURL -}} {{- $image := dict "@type" "ImageObject" "url" $imgURL -}} {{- $resource := $.Resources.GetMatch . -}} {{- if not $resource -}}{{- $resource = resources.Get (strings.TrimPrefix "/" .) -}}{{- end -}} {{- with $resource -}} {{- $preview := partialCached "seo/preview-image.html" . .Key -}} {{- $image = dict "@type" "ImageObject" "url" $preview.Permalink -}} {{- if ne $preview.MediaType.SubType "svg" -}} {{- $image = merge $image (dict "width" $preview.Width "height" $preview.Height) -}} {{- end -}} {{- end -}} {{- $schema = merge $schema (dict "image" $image) -}} {{- end -}} {{- /* keywords from tags (all types) */ -}} {{- with .Params.tags -}} {{- $schema = merge $schema (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- end -}}
