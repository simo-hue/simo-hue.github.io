{{- /* Schema.org structured data for blog posts. SEO-28: pick a more specific @type from the blog subsection (the 2nd URL segment of /blog///), overridable per-post via the front-matter field `schema_type`: tech-project -> SoftwareApplication publication -> ScholarlyArticle everything else (experience, thought, project, books) -> BlogPosting */ -}} {{- if and .IsPage (in site.Params.mainSections .Section) -}} {{- $desc := "" -}} {{- with .Description -}} {{- $desc = . -}} {{- else -}} {{- $desc = .Summary | plainify | truncate 160 -}} {{- end -}} {{- $author := site.Params.metadata.author -}} {{- with .Params.author -}} {{- $author = . -}} {{- end -}} {{- $personID := printf "%s#person" site.BaseURL -}} {{- /* subsection = 2nd path segment of the permalink */ -}} {{- $segs := split (strings.Trim .RelPermalink "/") "/" -}} {{- $sub := "" -}} {{- if ge (len $segs) 2 -}} {{- $sub = index $segs 1 -}} {{- end -}} {{- /* Match BOTH plural and singular. The real sections are plural (/blog/publications/, /blog/tech-projects/) while this only tested the singular forms, so the mapping never fired and every post — including the peer-reviewed article — fell back to BlogPosting. The singular forms are kept because each post also carries a singular alias. */ -}} {{- $type := "BlogPosting" -}} {{- if in (slice "tech-projects" "tech-project") $sub -}} {{- $type = "SoftwareApplication" -}} {{- else if in (slice "publications" "publication") $sub -}} {{- $type = "ScholarlyArticle" -}} {{- end -}} {{- with .Params.schema_type -}} {{- $type = . -}} {{- end -}} {{- /* properties common to every type */ -}} {{- $schema := dict "@context" "https://schema.org" "@type" $type "description" $desc "url" .Permalink "author" (dict "@type" "Person" "@id" $personID "name" $author) "datePublished" (.Date.Format "2006-01-02T15:04:05-07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05-07:00") "publisher" (dict "@type" "Person" "@id" $personID "name" site.Params.metadata.author "url" site.BaseURL) "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "inLanguage" "en-US" -}} {{- /* type-specific properties */ -}} {{- if eq $type "SoftwareApplication" -}} {{- $schema = merge $schema (dict "name" .Title "applicationCategory" (.Params.app_category | default "DeveloperApplication")) -}} {{- with .Params.operating_system -}} {{- $schema = merge $schema (dict "operatingSystem" .) -}} {{- end -}} {{- else -}} {{- $schema = merge $schema (dict "headline" .Title "wordCount" .WordCount) -}} {{- with .CurrentSection -}} {{- $schema = merge $schema (dict "articleSection" .Title) -}} {{- end -}} {{- end -}} {{- /* Bibliographic detail for real journal articles. Driven by front matter, so only posts that actually carry a DOI get these — the DOI is what lets Crossref, OpenAlex and Google Scholar reconcile this page with the published record rather than treating it as an unrelated blog post. */ -}} {{- if eq $type "ScholarlyArticle" -}} {{- with .Params.doi -}} {{- $schema = merge $schema (dict "@id" (printf "https://doi.org/%s" .) "identifier" (dict "@type" "PropertyValue" "propertyID" "DOI" "value" .) "sameAs" (printf "https://doi.org/%s" .)) -}} {{- end -}} {{- with .Params.authors -}} {{- $authors := slice -}} {{- range . -}} {{- $authors = $authors | append (dict "@type" "Person" "name" .) -}} {{- end -}} {{- /* Author order is meaningful: he is second of four, never sole author. */ -}} {{- $schema = merge $schema (dict "author" $authors) -}} {{- end -}} {{- with .Params.journal -}} {{- $periodical := dict "@type" "Periodical" "name" . -}} {{- with $.Params.publisher -}} {{- $periodical = merge $periodical (dict "publisher" (dict "@type" "Organization" "name" .)) -}} {{- end -}} {{- $schema = merge $schema (dict "isPartOf" $periodical) -}} {{- end -}} {{- with .Params.article_number -}} {{- $schema = merge $schema (dict "pagination" .) -}} {{- end -}} {{- end -}} {{- /* image (all types) */ -}} {{- with .Params.image -}} {{- $imgURL := . | absURL -}} {{- with $.Resources.GetMatch . -}} {{- $imgURL = .Permalink -}} {{- end -}} {{- $schema = merge $schema (dict "image" (dict "@type" "ImageObject" "url" $imgURL)) -}} {{- end -}} {{- /* keywords from tags (all types) */ -}} {{- with .Params.tags -}} {{- $schema = merge $schema (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- end -}}
